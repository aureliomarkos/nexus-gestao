CREATE TABLE public.clientes (
  id_cliente uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid NOT NULL,
  id_endereco uuid,
  nome character varying NOT NULL,
  email character varying NOT NULL UNIQUE,
  telefone character varying,
  pessoa_contato character varying,
  documento_fiscal character varying,
  segmento character varying,
  status_relacionamento character varying NOT NULL,
  origem character varying,
  observacoes text,
  data_criacao timestamp with time zone DEFAULT now(),
  data_ultimo_contato timestamp with time zone,
  CONSTRAINT clientes_pkey PRIMARY KEY (id_cliente),
  CONSTRAINT clientes_id_endereco_fkey FOREIGN KEY (id_endereco) REFERENCES public.enderecos(id_endereco)
);

  documento_fiscal character varying NOT NULL,
  tipo_contrato character varying,

  data_criacao timestamp with time zone DEFAULT now(),
  CONSTRAINT desenvolvedores_pkey PRIMARY KEY (id_desenvolvedor),
  CONSTRAINT desenvolvedores_id_endereco_fkey FOREIGN KEY (id_endereco) REFERENCES public.enderecos(id_endereco)
);
CREATE TABLE public.enderecos (
  id_endereco uuid NOT NULL DEFAULT gen_random_uuid(),

);
